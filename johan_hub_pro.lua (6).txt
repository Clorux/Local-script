local function isFiniteNumber(v)
    return type(v) == "number" and v == v and v > -math.huge and v < math.huge
end

local function createPropertyLock(getObject, propertyName, getTarget, fallbackDefault, minValid, maxValid)
    local minB = minValid or -math.huge
    local maxB = maxValid or math.huge
    local lock = { conn = nil, object = nil, baseline = fallbackDefault, weAreWriting = false }

    local function valuesClose(a, b)
        if type(a) == "number" and type(b) == "number" then
            return math.abs(a - b) <= 0.01
        end
        return a == b
    end

    local function sanitizeBaseline(val)
        if isFiniteNumber(val) and val >= minB and val <= maxB then
            return val
        end
        return fallbackDefault
    end

    function lock:install()
        self:uninstall()
        local success, obj = pcall(getObject)
        if not success or not obj then return end
        self.object = obj

        local gs, currentVal = pcall(function() return obj[propertyName] end)
        if not gs then return end
        local targetVal = getTarget()

        if not valuesClose(currentVal, targetVal) then
            self.baseline = sanitizeBaseline(currentVal)
        else
            self.baseline = fallbackDefault
        end

        self.conn = obj:GetPropertyChangedSignal(propertyName):Connect(function()
            if self.weAreWriting then
                self.weAreWriting = false
                return
            end
            local current = obj[propertyName]
            local t = getTarget()
            if not valuesClose(current, t) then
                if isFiniteNumber(current) and current >= minB and current <= maxB then
                    self.baseline = current
                end
                self.weAreWriting = true
                obj[propertyName] = t
            end
        end)

        if not valuesClose(obj[propertyName], targetVal) then
            self.weAreWriting = true
            obj[propertyName] = targetVal
        end
    end

    function lock:uninstall()
        if self.conn then
            self.conn:Disconnect()
            self.conn = nil
        end
        local obj, baseline = self.object, self.baseline
        self.object, self.baseline, self.weAreWriting = nil, nil, false
        if obj and baseline ~= nil then
            local alive, parentResult = pcall(function() return obj.Parent end)
            local valid = alive and ((obj == Workspace) or (obj == Lighting) or (parentResult ~= nil))
            if valid then
                pcall(function()
                    obj[propertyName] = baseline
                end)
            end
        end
    end

    function lock:refresh()
        if not self.object or not self.conn then return end
        local exists, current = pcall(function() return self.object[propertyName] end)
        if not exists then return end
        local t = getTarget()
        if not valuesClose(current, t) then
            self.weAreWriting = true
            self.object[propertyName] = t
        end
    end

    return lock
end

clearNoclipCacheConns = function()
    for _, conn in ipairs(noclipCharConns) do
        if conn.Connected then
            pcall(function() conn:Disconnect() end)
        end
    end
    table.clear(noclipCharConns)
    table.clear(noclipParts)

local function buildNoclipCache(char)
    clearNoclipCacheConns()
    if not char then return end

    for _, desc in ipairs(char:GetDescendants()) do
        if desc:IsA("BasePart") then
            table.insert(noclipParts, desc)
        end
    end

    table.insert(noclipCharConns, char.DescendantAdded:Connect(function(desc)
        if desc:IsA("BasePart") then
            table.insert(noclipParts, desc)
        end
    end))

    table.insert(noclipCharConns, char.DescendantRemoving:Connect(function(desc)
        if desc:IsA("BasePart") then
            local idx = table.find(noclipParts, desc)
            if idx then table.remove(noclipParts, idx) end
        end
    end))
end

local function restoreNoclip()
    local char = getCharacter()
    if not char then return end
    for _, part in ipairs(char:GetDescendants()) do
        if part:IsA("BasePart") then
            local original = part:GetAttribute("_originalCanCollide")
            if original ~= nil then
                part.CanCollide = original
                part:SetAttribute("_originalCanCollide", nil)
            end
        end
    end
end

startNoclipLoop = function()
    if noclipConnection then
        noclipConnection:Disconnect()
        noclipConnection = nil
    end
    buildNoclipCache(getCharacter())
    noclipConnection = RunService.Stepped:Connect(function()
        for i = 1, #noclipParts do
            local part = noclipParts[i]
            if part and part.Parent then
                if part:GetAttribute("_originalCanCollide") == nil then
                    part:SetAttribute("_originalCanCollide", part.CanCollide)
                end
                if part.CanCollide then
                    part.CanCollide = false
                end
            end
        end
    end)
end

stopNoclipLoop = function()
    if noclipConnection then
        noclipConnection:Disconnect()
        noclipConnection = nil
    end
    clearNoclipCacheConns()
    restoreNoclip()
end

local glideForcingNoclip = false

teleportToPosition = function(targetCFrame)
    local myRoot = getRootPart()
    if not myRoot then return end

    if activeGlideTween then
        pcall(function() activeGlideTween:Cancel() end)
        activeGlideTween = nil
        if glideForcingNoclip and not state.noclip then
            glideForcingNoclip = false
            stopNoclipLoop()
        end
    end

    if not state.smoothGlideTp then
        myRoot.CFrame = targetCFrame
        myRoot.AssemblyLinearVelocity = Vector3.zero
        return
    end

    local dist = (targetCFrame.Position - myRoot.Position).Magnitude
    if dist < 12 then
        myRoot.CFrame = targetCFrame
        return
    end

    local speed = math.clamp(state.glideSpeed or 135, 30, 400)
    local duration = math.clamp(dist / speed, 0.15, 8)
    if not state.noclip then
        glideForcingNoclip = true
        startNoclipLoop()
    end

    local glideTw = TweenService:Create(
        myRoot,
        TweenInfo.new(duration, Enum.EasingStyle.Linear),
        { CFrame = targetCFrame }
    )
    activeGlideTween = glideTw
    glideTw:Play()
    glideTw.Completed:Connect(function()
        if activeGlideTween == glideTw then
            activeGlideTween = nil
        end
        if myRoot and myRoot.Parent then
            myRoot.AssemblyLinearVelocity = Vector3.zero
        end
        if glideForcingNoclip then
            glideForcingNoclip = false
            if not state.noclip then
                stopNoclipLoop()
            end
        end
    end)
end

setAntiFling = function(enabled)
    if antiFlingConnection then
        antiFlingConnection:Disconnect()
        antiFlingConnection = nil
    end
    if not enabled then return end

    antiFlingConnection = RunService.Heartbeat:Connect(function()
        local hrp = getRootPart()
        if not hrp then return end

        local angVel = hrp.AssemblyAngularVelocity
        if angVel.Magnitude > 35 then
            hrp.AssemblyAngularVelocity = Vector3.zero
        end

        local linVel = hrp.AssemblyLinearVelocity
        local maxAllowed = math.max(180, (state.speed and state.speedValue or 24) * 2.2)
        if linVel.Magnitude > maxAllowed then
            hrp.AssemblyLinearVelocity = linVel.Unit * maxAllowed
        end
    end)
end

local function captureLightingBaseline()
    if savedLighting then return end
    savedLighting = {
        Brightness     = Lighting.Brightness,
        ClockTime      = Lighting.ClockTime,
        Ambient        = Lighting.Ambient,
        OutdoorAmbient = Lighting.OutdoorAmbient,
        GlobalShadows  = Lighting.GlobalShadows,
        FogEnd         = Lighting.FogEnd,
        FogStart       = Lighting.FogStart,
    }
    table.clear(savedAtmospheres)
    for _, child in ipairs(Lighting:GetChildren()) do
        if child:IsA("Atmosphere") then
            savedAtmospheres[child] = child.Density
        end
    end
end

syncLightingOverrides = function()
    clearLightingLockConns()

    if not state.fullbright and not state.noFog then
        if savedLighting then
            pcall(function()
                Lighting.Brightness     = savedLighting.Brightness
                Lighting.ClockTime      = savedLighting.ClockTime
                Lighting.Ambient        = savedLighting.Ambient
                Lighting.OutdoorAmbient = savedLighting.OutdoorAmbient
                Lighting.GlobalShadows  = savedLighting.GlobalShadows
                Lighting.FogEnd         = savedLighting.FogEnd
                Lighting.FogStart       = savedLighting.FogStart
            end)
            for atm, density in pairs(savedAtmospheres) do
                if atm and atm.Parent then
                    pcall(function() atm.Density = density end)
                end
            end
            table.clear(savedAtmospheres)
            savedLighting = nil
        end
        return
    end

    captureLightingBaseline()

    local function applyNow()
        if state.fullbright then
            Lighting.Brightness     = 2.2
            Lighting.ClockTime      = 14
            Lighting.Ambient        = Color3.fromRGB(178, 178, 178)
            Lighting.OutdoorAmbient = Color3.fromRGB(178, 178, 178)
            Lighting.GlobalShadows  = false
        elseif savedLighting then
            Lighting.Brightness     = savedLighting.Brightness
            Lighting.ClockTime      = savedLighting.ClockTime
            Lighting.Ambient        = savedLighting.Ambient
            Lighting.OutdoorAmbient = savedLighting.OutdoorAmbient
            Lighting.GlobalShadows  = savedLighting.GlobalShadows
        end

        if state.noFog then
            Lighting.FogEnd   = 100000
            Lighting.FogStart = 100000
            for _, child in ipairs(Lighting:GetChildren()) do
                if child:IsA("Atmosphere") then
                    if savedAtmospheres[child] == nil then
                        savedAtmospheres[child] = child.Density
                    end
                    child.Density = 0
                end
            end
        elseif savedLighting then
            Lighting.FogEnd   = savedLighting.FogEnd
            Lighting.FogStart = savedLighting.FogStart
            for atm, density in pairs(savedAtmospheres) do
                if atm and atm.Parent then
                    pcall(function() atm.Density = density end)
                end
            end
        end
    end

    local isApplyingLighting = false
    local function guardedApply()
        if isApplyingLighting then return end
        isApplyingLighting = true
        pcall(applyNow)
        isApplyingLighting = false
    end

    guardedApply()

    for _, prop in ipairs({ "Brightness", "ClockTime", "Ambient", "OutdoorAmbient", "GlobalShadows", "FogEnd", "FogStart" }) do
        table.insert(lightingLockConns, Lighting:GetPropertyChangedSignal(prop):Connect(guardedApply))
    end
    table.insert(lightingLockConns, Lighting.ChildAdded:Connect(function(child)
        if child:IsA("Atmosphere") then
            task.defer(guardedApply)
        end
    end))
end
end

local function drawCached2DLine(lineObj, ax, ay, bx, by, color, thickness, transparency)
    local dx = bx - ax
    local dy = by - ay
    local length = math.sqrt(dx * dx + dy * dy)
    if length < 1.0 then
        if lineObj.vis then
            lineObj.vis = false
            lineObj.inst.Visible = false
        end
        return
    end

    local midX = (ax + bx) * 0.5
    local midY = (ay + by) * 0.5
    local rot  = math.deg(math.atan2(dy, dx))
    local inst = lineObj.inst
    local th   = thickness or 1

    if math.abs(midX - lineObj.x) >= 0.45 or math.abs(midY - lineObj.y) >= 0.45 then
        lineObj.x = midX
        lineObj.y = midY
        inst.Position = UDim2.fromOffset(midX, midY)
    end

    if math.abs(length - lineObj.len) >= 0.45 or th ~= lineObj.thick then
        lineObj.len   = length
        lineObj.thick = th
        inst.Size     = UDim2.fromOffset(length, th)
    end

    if math.abs(rot - lineObj.rot) >= 0.35 then
        lineObj.rot   = rot
        inst.Rotation = rot
    end

    if lineObj.color ~= color then
        lineObj.color         = color
        inst.BackgroundColor3 = color
    end

    if transparency and math.abs(transparency - lineObj.trans) >= 0.02 then
        lineObj.trans               = transparency
        inst.BackgroundTransparency = transparency
    end

    if not lineObj.vis then
        lineObj.vis  = true
        inst.Visible = true
    end
end

FrameScheduler = {
    emaDt                   = 0.0166,
    tier                    = 1,
    frameCounter            = 0,
    lodTier1Dist            = 160,
    rayBudgetRemaining      = 8,
    losInterval             = 0.10,
    allowSecondaryThisFrame = true,
}

function FrameScheduler:beginFrame(dt)
    self.frameCounter = self.frameCounter + 1
    local clampedDt = math.clamp(dt, 0.001, 0.25)
    local alpha = (clampedDt > self.emaDt) and 0.25 or 0.05
    local ema = self.emaDt + (clampedDt - self.emaDt) * alpha
    self.emaDt = ema

    if self.tier == 1 then
        if ema > 0.0190 then self.tier = 2 end
    elseif self.tier == 2 then
        if ema < 0.0172 then
            self.tier = 1
        elseif ema > 0.0245 then
            self.tier = 3
        end
    else
        if ema < 0.0215 then self.tier = 2 end
    end

    if self.tier == 1 then
        self.lodTier1Dist            = 160
        self.rayBudgetRemaining      = 8
        self.losInterval             = 0.10
        self.allowSecondaryThisFrame = true
    elseif self.tier == 2 then
        self.lodTier1Dist            = 105
        self.rayBudgetRemaining      = 4
        self.losInterval             = 0.18
        self.allowSecondaryThisFrame = (self.frameCounter % 2 == 0)
    else
        self.lodTier1Dist            = 55
        self.rayBudgetRemaining      = 2
        self.losInterval             = 0.30
        self.allowSecondaryThisFrame = (self.frameCounter % 4 == 0)
    end
end

function FrameScheduler:consumeRayToken()
    if self.rayBudgetRemaining > 0 then
        self.rayBudgetRemaining = self.rayBudgetRemaining - 1
        return true
    end
    return false
end

local FrustumCuller = {
    camPos   = Vector3.zero,
    camRight = Vector3.xAxis,
    camUp    = Vector3.yAxis,
    camLook  = -Vector3.zAxis,
    tanHalfY = 0.7002,
    tanHalfX = 1.2448,
    secHalfY = 1.2207,
    secHalfX = 1.5961,
    vpHalfX  = 960,
    vpHalfY  = 540,
    focalX   = 771.2,
    focalY   = 771.2,
    nearZ    = 0.25,
}

function FrustumCuller:updateFromCamera(cam)
    local cf     = cam.CFrame
    local vp     = cam.ViewportSize
    local vx     = math.max(1, vp.X)
    local vy     = math.max(1, vp.Y)
    local aspect = vx / vy
    local halfY  = math.rad(cam.FieldOfView * 0.5)
    local ty     = math.tan(halfY)
    local tx     = ty * aspect

    self.camPos   = cf.Position
    self.camRight = cf.RightVector
    self.camUp    = cf.UpVector
    self.camLook  = cf.LookVector
    self.tanHalfY = ty
    self.tanHalfX = tx
    self.secHalfY = math.sqrt(1 + ty * ty)
    self.secHalfX = math.sqrt(1 + tx * tx)
    self.vpHalfX  = vx * 0.5
    self.vpHalfY  = vy * 0.5
    self.focalY   = (vy * 0.5) / math.max(1e-5, ty)
    self.focalX   = (vx * 0.5) / math.max(1e-5, tx)
end

function FrustumCuller:testSphere(worldPos, radius, maxDist)
    local rel    = worldPos - self.camPos
    local depthZ = rel:Dot(self.camLook)
    if depthZ + radius < self.nearZ or depthZ - radius > maxDist then
        return false, 0, 0, depthZ
    end

    local camY = rel:Dot(self.camUp)
    if math.abs(camY) > depthZ * self.tanHalfY + radius * self.secHalfY then
        return false, 0, camY, depthZ
    end

    local camX = rel:Dot(self.camRight)
    if math.abs(camX) > depthZ * self.tanHalfX + radius * self.secHalfX then
        return false, camX, camY, depthZ
    end

    return true, camX, camY, depthZ
end

function FrustumCuller:projectClippedSegment(ptA, ptB)
    local relA = ptA - self.camPos
    local relB = ptB - self.camPos
    local zA   = relA:Dot(self.camLook)
    local zB   = relB:Dot(self.camLook)
    local near = self.nearZ

    if zA < near and zB < near then
        return false, 0, 0, 0, 0
    end

    local xA, yA = relA:Dot(self.camRight), relA:Dot(self.camUp)
    local xB, yB = relB:Dot(self.camRight), relB:Dot(self.camUp)

    if zA < near then
        local t = (near - zA) / math.max(1e-5, zB - zA)
        xA = xA + (xB - xA) * t
        yA = yA + (yB - yA) * t
        zA = near
    elseif zB < near then
        local t = (near - zB) / math.max(1e-5, zA - zB)
        xB = xB + (xA - xB) * t
        yB = yB + (yA - yB) * t
        zB = near
    end

    local invZA = 1.0 / zA
    local invZB = 1.0 / zB
    local sxA   = self.vpHalfX + xA * invZA * self.focalX
    local syA   = self.vpHalfY - yA * invZA * self.focalY
    local sxB   = self.vpHalfX + xB * invZB * self.focalX
    local syB   = self.vpHalfY - yB * invZB * self.focalY

    return true, sxA, syA, sxB, syB
end

function FrustumCuller:projectPointClipped(worldPt)
    local rel = worldPt - self.camPos
    local z   = math.max(self.nearZ, rel:Dot(self.camLook))
    local x   = rel:Dot(self.camRight)
    local y   = rel:Dot(self.camUp)
    local invZ = 1.0 / z
    return self.vpHalfX + x * invZ * self.focalX, self.vpHalfY - y * invZ * self.focalY, rel:Dot(self.camLook) >= self.nearZ
end

local stepEntityKinematics
do
local ALPHA_GAIN  = 0.65
local BETA_V_GAIN = 0.45
local BETA_P_GAIN = 0.12
local GAMMA_GAIN  = 0.18
local GAMMA_DECAY = 0.08
local MAX_OMEGA_Y = 4.5

stepEntityKinematics = function(slot, now, rawPos, rawVel, leadTime)
    local base     = (slot - 1) * ENTITY_STRIDE
    local lastTick = BufferLib.readf32(entitySoABuffer, base + 52)
    BufferLib.writef32(entitySoABuffer, base + 52, now)

    if lastTick <= 0 or (now - lastTick) > 0.5 then
        BufferLib.writef32(entitySoABuffer, base + 0, rawPos.X)
        BufferLib.writef32(entitySoABuffer, base + 4, rawPos.Y)
        BufferLib.writef32(entitySoABuffer, base + 8, rawPos.Z)
        BufferLib.writef32(entitySoABuffer, base + 12, rawVel.X)
        BufferLib.writef32(entitySoABuffer, base + 16, rawVel.Y)
        BufferLib.writef32(entitySoABuffer, base + 20, rawVel.Z)
        BufferLib.writef32(entitySoABuffer, base + 24, 0)
        BufferLib.writef32(entitySoABuffer, base + 28, 0)
        BufferLib.writef32(entitySoABuffer, base + 32, 0)
        BufferLib.writef32(entitySoABuffer, base + 36, 0)
        return rawPos + rawVel * leadTime, rawVel, Vector3.zero, 0
    end

    local dt = math.clamp(now - lastTick, 0.001, 0.25)
    local invDt = 1.0 / dt
    local halfDtSq = 0.5 * dt * dt

    local px = BufferLib.readf32(entitySoABuffer, base + 0)
    local py = BufferLib.readf32(entitySoABuffer, base + 4)
    local pz = BufferLib.readf32(entitySoABuffer, base + 8)
    local vx = BufferLib.readf32(entitySoABuffer, base + 12)
    local vy = BufferLib.readf32(entitySoABuffer, base + 16)
    local vz = BufferLib.readf32(entitySoABuffer, base + 20)
    local ax = BufferLib.readf32(entitySoABuffer, base + 24)
    local ay = BufferLib.readf32(entitySoABuffer, base + 28)
    local az = BufferLib.readf32(entitySoABuffer, base + 32)

    local predPx = px + vx * dt + ax * halfDtSq
    local predPy = py + vy * dt + ay * halfDtSq
    local predPz = pz + vz * dt + az * halfDtSq
    local predVx = vx + ax * dt
    local predVy = vy + ay * dt
    local predVz = vz + az * dt

    local resPx = rawPos.X - predPx
    local resPy = rawPos.Y - predPy
    local resPz = rawPos.Z - predPz
    local resVx = rawVel.X - predVx
    local resVy = rawVel.Y - predVy
    local resVz = rawVel.Z - predVz

    px = predPx + ALPHA_GAIN * resPx
    py = predPy + ALPHA_GAIN * resPy
    pz = predPz + ALPHA_GAIN * resPz

    local bpFactor = BETA_P_GAIN * invDt
    vx = predVx + BETA_V_GAIN * resVx + bpFactor * resPx
    vy = predVy + BETA_V_GAIN * resVy + bpFactor * resPy
    vz = predVz + BETA_V_GAIN * resVz + bpFactor * resPz

    local gFactor = GAMMA_GAIN * invDt
    local decay   = 1.0 - GAMMA_DECAY
    ax = ax * decay + gFactor * resVx
    ay = ay * decay + gFactor * resVy
    az = az * decay + gFactor * resVz

    local speedSqXZ = vx * vx + vz * vz
    local omegaY = 0
    if speedSqXZ > 4.0 then
        local rawOmega = (vz * ax - vx * az) / speedSqXZ
        local prevOmega = BufferLib.readf32(entitySoABuffer, base + 36)
        omegaY = math.clamp(prevOmega + (rawOmega - prevOmega) * 0.35, -MAX_OMEGA_Y, MAX_OMEGA_Y)
    end

    BufferLib.writef32(entitySoABuffer, base + 0, px)
    BufferLib.writef32(entitySoABuffer, base + 4, py)
    BufferLib.writef32(entitySoABuffer, base + 8, pz)
    BufferLib.writef32(entitySoABuffer, base + 12, vx)
    BufferLib.writef32(entitySoABuffer, base + 16, vy)
    BufferLib.writef32(entitySoABuffer, base + 20, vz)
    BufferLib.writef32(entitySoABuffer, base + 24, ax)
    BufferLib.writef32(entitySoABuffer, base + 28, ay)
    BufferLib.writef32(entitySoABuffer, base + 32, az)
    BufferLib.writef32(entitySoABuffer, base + 36, omegaY)

    local futX, futZ
    if math.abs(omegaY) > 0.08 and speedSqXZ > 6.0 then
        local theta = omegaY * leadTime
        local sinT  = math.sin(theta)
        local cosT  = math.cos(theta)
        local invW  = 1.0 / omegaY
        futX = rawPos.X + (vx * sinT + vz * (1.0 - cosT)) * invW
        futZ = rawPos.Z + (vz * sinT - vx * (1.0 - cosT)) * invW
    else
        futX = rawPos.X + vx * leadTime + 0.5 * ax * (leadTime * leadTime)
        futZ = rawPos.Z + vz * leadTime + 0.5 * az * (leadTime * leadTime)
    end
    local futY = rawPos.Y + vy * 0.25

    return Vector3.new(futX, futY, futZ), Vector3.new(vx, vy, vz), Vector3.new(ax, ay, az), omegaY
end

end
local function computeGroundClampedBallisticArc(originPos, smoothVel, gravity, rayParams, allowRaycast)
    local g = math.max(1, gravity)
    local vy = smoothVel.Y
    local disc = vy * vy + 2.0 * g * 32.0
    local maxFlightT = math.clamp((vy + math.sqrt(math.max(0, disc))) / g, 0.25, 1.40)
    local stepT = maxFlightT * 0.25

    scratchBallisticPts[1] = originPos
    local validCount = 1
    local impactWorld = nil

    for seg = 1, 4 do
        local t = seg * stepT
        local nextPt = Vector3.new(
            originPos.X + smoothVel.X * t,
            originPos.Y + vy * t - 0.5 * g * (t * t),
            originPos.Z + smoothVel.Z * t
        )
        local prevPt = scratchBallisticPts[seg]
        if allowRaycast and nextPt.Y <= prevPt.Y then
            local hit = Workspace:Raycast(prevPt, nextPt - prevPt, rayParams)
            if hit then
                validCount = validCount + 1
                scratchBallisticPts[validCount] = hit.Position
                impactWorld = hit.Position
                break
            end
        end
        validCount = validCount + 1
        scratchBallisticPts[validCount] = nextPt
        impactWorld = nextPt
    end

    return validCount, impactWorld
end

    node.maxX   = centerX + hw
    node.weight = priorityWeight
end

solveAndCommitSapLabels = function(viewportHeight)
    local n = sapActiveCount
    if n <= 1 then
        if n == 1 then
            local node = sapPool[1]
            node.frame.Position = UDim2.fromOffset(math.floor(node.cx + 0.5), math.floor(node.cy + 0.5))
        end
        return
    end

    for i = 2, n do
        local key = sapPool[i]
        local keyMinX = key.minX
        local j = i - 1
        while j >= 1 and sapPool[j].minX > keyMinX do
            sapPool[j + 1] = sapPool[j]
            j = j - 1
        end
        sapPool[j + 1] = key
    end

    for _ = 1, 2 do
        for i = 1, n - 1 do
            local a = sapPool[i]
            local aMaxX = a.maxX
            for j = i + 1, n do
                local b = sapPool[j]
                if b.minX >= aMaxX then
                    break
                end
                local reqSepY = a.halfH + b.halfH + 2.0
                local dy = b.cy - a.cy
                local absDy = math.abs(dy)
                if absDy < reqSepY then
                    local overlap = reqSepY - absDy
                    local dir = (dy >= 0) and 1.0 or -1.0
                    local totalW = a.weight + b.weight
                    local ratioA = b.weight / totalW
                    local ratioB = a.weight / totalW
                    a.cy = math.clamp(a.cy - dir * overlap * ratioA, 14, viewportHeight - 14)
                    b.cy = math.clamp(b.cy + dir * overlap * ratioB, 14, viewportHeight - 14)
                end
            end
        end
    end

    for i = 1, n do
        local node = sapPool[i]
        node.frame.Position = UDim2.fromOffset(math.floor(node.cx + 0.5), math.floor(node.cy + 0.5))
    end
end

end

local ACTION_KEYWORDS = {
    { token = "reload",  badge = "RELOADING", priority = 10 },
    { token = "heal",    badge = "HEALING",   priority = 9 },
    { token = "bandage", badge = "HEALING",   priority = 9 },

    { token = "parry",   badge = "PARRYING",  priority = 7 },
    { token = "aim",     badge = "ADS / AIM", priority = 6 },

classifyTargetActionState = function(entry, now, char, hum, flatSpeed)
    local hState = hum:GetState()
    if hState == Enum.HumanoidStateType.Climbing then
        return "CLIMBING"
    elseif hState == Enum.HumanoidStateType.Swimming then
        return "SWIMMING"
    elseif hState == Enum.HumanoidStateType.FallingDown or hState == Enum.HumanoidStateType.Ragdoll then
        return "RAGDOLLED"
    elseif hum.Sit then
        local seat = hum.SeatPart
        if seat and seat:IsA("VehicleSeat") then
            return "DRIVING"
        end
        return "SEATED"
    end

    if now >= (entry.nextAnimScanAt or 0) then
        entry.nextAnimScanAt = now + 0.15
        local detectedBadge = nil
        local bestPrio = 0
        local animator = hum:FindFirstChildOfClass("Animator")
        if animator then
            local ok, tracks = pcall(function() return animator:GetPlayingAnimationTracks() end)
            if ok and tracks then
                for i = 1, #tracks do
                    local tr = tracks[i]
                    if tr.IsPlaying and tr.WeightCurrent > 0.2 then
                        local anim = tr.Animation
                        local rawName = string.lower((tr.Name or "") .. " " .. (anim and anim.Name or ""))
                        for k = 1, #ACTION_KEYWORDS do
                            local rule = ACTION_KEYWORDS[k]
                            if rule.priority > bestPrio and string.find(rawName, rule.token, 1, true) then
                                detectedBadge = rule.badge
                                bestPrio = rule.priority
                            end
                        end
                        if not detectedBadge and tr.Priority == Enum.AnimationPriority.Action4 then
                            detectedBadge = "ABILITY"
                        end
                    end
                end
            end
        end
        entry.cachedActionBadge = detectedBadge
    end

    if entry.cachedActionBadge then
        return entry.cachedActionBadge
    end

    if hState == Enum.HumanoidStateType.Jumping or hState == Enum.HumanoidStateType.Freefall then
        return "AIRBORNE"
    elseif flatSpeed > 22 then
        return string.format("SPRINT (%d)", math.floor(flatSpeed + 0.5))
    elseif flatSpeed > 2 then
        return string.format("MOVE (%d)", math.floor(flatSpeed + 0.5))
    end
    return "IDLE"
end

end
local function pushEntityTrailSample(slot, now, pos, lastSampleAt)
    if now - lastSampleAt < 0.10 then
        return lastSampleAt
    end
    local base    = (slot - 1) * ENTITY_STRIDE
    local headIdx = BufferLib.readu8(entitySoABuffer, base + 56)
    local count   = BufferLib.readu8(entitySoABuffer, base + 57)

    if count > 0 then
        local prevOffset = base + TRAIL_BASE_OFFSET + ((headIdx - 1) * 12)
        local dx = pos.X - BufferLib.readf32(entitySoABuffer, prevOffset)
        local dy = pos.Y - BufferLib.readf32(entitySoABuffer, prevOffset + 4)
        local dz = pos.Z - BufferLib.readf32(entitySoABuffer, prevOffset + 8)
        if (dx * dx + dy * dy + dz * dz) < 1.44 then
            return lastSampleAt
        end
    end

    local nextHead = (headIdx % TRAIL_CAP) + 1
    local writeOff = base + TRAIL_BASE_OFFSET + ((nextHead - 1) * 12)
    BufferLib.writef32(entitySoABuffer, writeOff, pos.X)
    BufferLib.writef32(entitySoABuffer, writeOff + 4, pos.Y)
    BufferLib.writef32(entitySoABuffer, writeOff + 8, pos.Z)
    BufferLib.writeu8(entitySoABuffer, base + 56, nextHead)
    if count < TRAIL_CAP then
        BufferLib.writeu8(entitySoABuffer, base + 57, count + 1)
    end
    return now
end

local function evaluateAimConeSeverity(enemyChar, enemyHrp, myOriginPos, dist, isVisible)
    if not state.espAimAlert or dist > 220 or dist < 0.5 then
        return 0, 999
    end
    local head = enemyChar:FindFirstChild("Head")
    local origin = (head and head:IsA("BasePart")) and head.Position or enemyHrp.Position
    local toMe = myOriginPos - origin
    local d = toMe.Magnitude
    if d < 0.5 then return 0, 999 end
    local unitToMe = toMe / d

    local lookHead = (head and head:IsA("BasePart")) and head.CFrame.LookVector or enemyHrp.CFrame.LookVector
    local maxDot   = lookHead:Dot(unitToMe)

    local tool = enemyChar:FindFirstChildOfClass("Tool")
    local handle = tool and tool:FindFirstChild("Handle")
    if handle and handle:IsA("BasePart") then
        local toolDot = handle.CFrame.LookVector:Dot(unitToMe)
        if toolDot > maxDot then maxDot = toolDot end
    end

    if maxDot < 0.885 then
        return 0, 999
    end

    local sinSq    = math.max(0, 1.0 - maxDot * maxDot)
    local missDist = d * math.sqrt(sinSq)

    if maxDot >= 0.965 and missDist <= 4.5 and (isVisible or dist <= 55) then
        return 2, missDist
    elseif missDist <= 11.0 and (isVisible or dist <= 85) then
        return 1, missDist
    end
    return 0, missDist
end

    if entry.rootFrame.Visible then entry.rootFrame.Visible = false end
    hideCachedLine(entry.tracerLine)
    hideCachedLine(entry.lookRayLine)
    hideCachedLine(entry.velLine)
    if entry.velDot.Visible then entry.velDot.Visible = false end
    if entry.headDot.Visible then entry.headDot.Visible = false end
    if entry.arrowFrame.Visible then entry.arrowFrame.Visible = false end
    if entry.radarBlip.Visible then entry.radarBlip.Visible = false end
    if entry.highlight.Enabled then entry.highlight.Enabled = false end
    if entry.landingDot.Visible then entry.landingDot.Visible = false end
    for i = 1, #entry.boneLines do hideCachedLine(entry.boneLines[i]) end
    for i = 1, #entry.box3DLines do hideCachedLine(entry.box3DLines[i]) end
    for i = 1, #entry.footstepLines do hideCachedLine(entry.footstepLines[i]) end
    for i = 1, #entry.arcLines do hideCachedLine(entry.arcLines[i]) end
    for i = 1, #entry.trailLines do hideCachedLine(entry.trailLines[i]) end
end

local function isSameTeam(player)
    if not state.espTeamCheck then return false end
    if not LocalPlayer.Team or not player.Team then return false end
    return LocalPlayer.Team == player.Team
end

local function getCachedVisibility(entry, now, cam, targetChar, targetPart)
    if not state.espWallCheck then return true end
    if now < entry.nextLosAt then
        return entry.cachedVisible
    end
    if not FrameScheduler:consumeRayToken() then
        return entry.cachedVisible
    end
    entry.nextLosAt = now + FrameScheduler.losInterval
    local myChar = getCharacter()
    losRayParams.FilterDescendantsInstances = myChar and { myChar, cam } or { cam }
    local origin = cam.CFrame.Position
    local dir = targetPart.Position - origin
    if dir.Magnitude < 2 then
        entry.cachedVisible = true
        return true
    end
    local hit = Workspace:Raycast(origin, dir, losRayParams)
    local vis = (not hit or not hit.Instance) or hit.Instance:IsDescendantOf(targetChar)
    entry.cachedVisible = vis
    return vis
end

local function getCachedHalfExtents(entry, now, char)
    if now < entry.nextExtentsAt then
        return entry.cachedHalfExt
    end
    entry.nextExtentsAt = now + 1.5
    local ok, sz = pcall(function() return char:GetExtentsSize() end)
    if ok and sz then
        entry.cachedHalfExt = Vector3.new(
            math.clamp(sz.X * 0.5, 1.4, 4.5),
            math.clamp(sz.Y * 0.5, 2.2, 5.5),
            math.clamp(sz.Z * 0.5, 0.9, 3.5)
        )
    end
    return entry.cachedHalfExt
end

local function getPlayerEspColor(player, toolName, dist, isVisible, aimSeverity)
    if aimSeverity == 2 then
        return Theme.Danger
    elseif aimSeverity == 1 then
        return Theme.WarnAmber
    end
    if player.Team and player.TeamColor then
        return player.TeamColor.Color
    end
    if toolName and dist < 85 then
        return Theme.Danger
    end
    if state.espWallCheck then
        return isVisible and Theme.SafeGreen or Theme.OccludedIndigo
    end
    return Theme.Accent
end

local function getStreamingSafeBoneParts(entry, now, char, hum)
    local needsRebuild = (entry.cachedBoneChar ~= char) or (now >= (entry.nextBoneValidateAt or 0))
    if needsRebuild then
        entry.cachedBoneChar = char
        entry.nextBoneValidateAt = now + 0.50
        local boneList = (hum.RigType == Enum.HumanoidRigType.R15) and R15_BONES or R6_BONES
        for i = 1, #boneList do
            local existingPair = entry.cachedBonePairs[i]
            local validPair = existingPair
                and existingPair[1] and existingPair[1].Parent == char
                and existingPair[2] and existingPair[2].Parent == char
            if not validPair then
                local pair = boneList[i]
                local pA = char:FindFirstChild(pair[1])
                local pB = char:FindFirstChild(pair[2])
                if pA and pB and pA:IsA("BasePart") and pB:IsA("BasePart") then
                    entry.cachedBonePairs[i] = { pA, pB }
                else
                    entry.cachedBonePairs[i] = false
                end
            end
        end
    end
    return entry.cachedBonePairs
end

local function stepEntityHealthAndPopups(entry, now, hum, onScreen, screenX, screenY)
    local safeMaxHp  = (isFiniteNumber(hum.MaxHealth) and hum.MaxHealth > 0) and hum.MaxHealth or 100
    local safeHealth = isFiniteNumber(hum.Health) and math.clamp(hum.Health, 0, safeMaxHp * 10) or safeMaxHp

    if entry.lastHpValue ~= nil and isFiniteNumber(entry.lastHpValue) then
        local hpDiff = safeHealth - entry.lastHpValue
        if math.abs(hpDiff) >= 0.5 and math.abs(hpDiff) < 1e6 then
            entry.dmgAccumulator = (isFiniteNumber(entry.dmgAccumulator) and entry.dmgAccumulator or 0) + hpDiff
        end
        if math.abs(entry.dmgAccumulator) >= 1 and (now - entry.lastDmgPopupAt) >= 0.12 then
            if state.espDamagePopups and onScreen then
                spawnDamagePopup(screenX, screenY - 22, entry.dmgAccumulator)
            end
            entry.dmgAccumulator = 0
            entry.lastDmgPopupAt = now
        end
    else
        entry.dmgAccumulator = 0
    end
    entry.lastHpValue = safeHealth
    return safeHealth, safeMaxHp
end

    if not state.espTargetHud or not bestPlayer or not bestChar or not bestHum or not bestHrp then
        PiPWidgetManager:syncVisibility("TargetHUD", false)
        return
    end

    PiPWidgetManager:syncVisibility("TargetHUD", true)
    targetHudStroke.Color = lockedTargetPlayer == bestPlayer and Theme.WarnAmber or bestColor

    if lastTargetHudUserId ~= bestPlayer.UserId then
        lastTargetHudUserId = bestPlayer.UserId
        targetAvatar.Image = string.format("rbxthumb://type=AvatarHeadShot&id=%d&w=150&h=150", bestPlayer.UserId)
    end

    local lockPrefix = (lockedTargetPlayer == bestPlayer) and "[PINNED] " or ""
    targetHudName.Text = lockPrefix .. (bestPlayer.DisplayName ~= bestPlayer.Name
        and string.format("%s (@%s)", bestPlayer.DisplayName, bestPlayer.Name)

    local safeHp  = isFiniteNumber(bestHum.Health) and math.clamp(bestHum.Health, 0, safeMax) or safeMax

    targetHudHpFill.Size = UDim2.fromScale(ratio, 1)
    targetHudHpFill.BackgroundColor3 = hpColor
    targetHudHpLabel.Text = string.format("%d / %d HP (%d%%)", math.ceil(safeHp), math.ceil(safeMax), math.floor(ratio * 100 + 0.5))

SmartObjectScanner = {
    registry      = {},
    activeList    = table.create(128),
    whitelist     = {},
    conns         = {},
    scanCoroutine = nil,
    keywords      = {
        "chest", "crate", "drop", "key", "gem", "orb", "vault",
        "safe", "loot", "artifact", "relic", "briefcase", "shard", "money", "cash",
    },
    attrTokens    = { "rarity", "loot", "value", "tier", "collectible", "itemid", "chest" },
}

function SmartObjectScanner:resolveBasePart(inst)
    if inst:IsA("BasePart") then return inst end
    if inst:IsA("Model") then
        return inst.PrimaryPart or inst:FindFirstChildWhichIsA("BasePart", true)
    end
    local ancPart = inst:FindFirstAncestorWhichIsA("BasePart")
    if ancPart then return ancPart end
    local ancModel = inst:FindFirstAncestorOfClass("Model")
    if ancModel then
        return ancModel.PrimaryPart or ancModel:FindFirstChildWhichIsA("BasePart", true)
    end
    return nil
end

function SmartObjectScanner:classifyInstance(inst, myChar)
    if not inst or not inst.Parent then return end
    if myChar and inst:IsDescendantOf(myChar) then return end
    if self.registry[inst] then return end

    local ancModel = inst:FindFirstAncestorOfClass("Model")
    if ancModel and ancModel:FindFirstChildOfClass("Humanoid") then return end

    local score = 0
    local tag   = nil
    local color = Theme.SafeGreen

    if inst:IsA("ProximityPrompt") and inst.Enabled then
        local host = inst.Parent
        local label = (inst.ObjectText ~= "" and inst.ObjectText)
            or (inst.ActionText ~= "" and inst.ActionText)
            or (host and host.Name) or "Interact"
        score = 100
        tag   = "[PROMPT] " .. label
        color = Theme.SafeGreen
    elseif inst:IsA("ClickDetector") then
        local host = inst.Parent
        score = 85
        tag   = "[CLICK] " .. (host and host.Name or "Object")
        color = Theme.SafeGreen
    elseif inst:IsA("Tool") then
        score = 95
        tag   = "[TOOL] " .. inst.Name
        color = Theme.WarnAmber
    elseif inst:IsA("Model") or inst:IsA("BasePart") then
        local lowName = string.lower(inst.Name)
        if self.whitelist[lowName] then
            score = 110
            tag   = "[WL] " .. inst.Name
            color = Theme.WaypointCyan
        else
            for i = 1, #self.keywords do
                if string.find(lowName, self.keywords[i], 1, true) then
                    score = 80
                    tag   = "[ITEM] " .. inst.Name
                    color = Theme.WaypointCyan
                    break
                end
            end
            if score == 0 then
                local ok, attrs = pcall(function() return inst:GetAttributes() end)
                if ok and attrs then
                    for k, _ in pairs(attrs) do
                        local lowK = string.lower(k)
                        for a = 1, #self.attrTokens do
                            if string.find(lowK, self.attrTokens[a], 1, true) then
                                score = 90
                                tag   = "[LOOT] " .. inst.Name
                                color = Theme.WaypointCyan
                                break
                            end
                        end
                        if score > 0 then break end
                    end
                end
            end
        end
    end

    if score > 0 and tag then
        local rootPart = self:resolveBasePart(inst)
        if rootPart then
            local rec = {
                inst      = inst,
                rootPart  = rootPart,
                tag       = tag,
                color     = color,
                baseScore = score,
            }
            self.registry[inst] = rec
            table.insert(self.activeList, rec)
        end
    end
end

function SmartObjectScanner:removeInstance(inst)
    local rec = self.registry[inst]
    if not rec then return end
    self.registry[inst] = nil
    local idx = table.find(self.activeList, rec)
    if idx then
        local last = #self.activeList
        self.activeList[idx] = self.activeList[last]
        self.activeList[last] = nil
    end
end

function SmartObjectScanner:start()
    self:stop()
    table.insert(self.conns, Workspace.DescendantAdded:Connect(function(desc)
        self:classifyInstance(desc, getCharacter())
    end))
    table.insert(self.conns, Workspace.DescendantRemoving:Connect(function(desc)
        self:removeInstance(desc)
    end))

    self.scanCoroutine = task.spawn(function()
        local descendants = Workspace:GetDescendants()
        local myChar = getCharacter()
        for i = 1, #descendants do
            self:classifyInstance(descendants[i], myChar)
            if i % 1200 == 0 then
                task.wait()
                myChar = getCharacter()
            end
        end
    end)
end

function SmartObjectScanner:stop()
    if self.scanCoroutine then
        pcall(task.cancel, self.scanCoroutine)
        self.scanCoroutine = nil
    end
    for i = 1, #self.conns do
        if self.conns[i].Connected then
            self.conns[i]:Disconnect()
        end
    end
    table.clear(self.conns)
    table.clear(self.registry)
    table.clear(self.activeList)
end

local function stepWorldObjectsESP(cam, originPos)
    if not state.worldObjectEsp then
        hideAllObjectEspPool()
        return
    end
    local maxObjDist = state.objectEspMaxDist or 350
    local used = 0
    local list = SmartObjectScanner.activeList
    for i = 1, #list do
        if used >= OBJECT_POOL_SIZE then break end
        local item = list[i]
        local rp = item.rootPart
        if rp and rp.Parent then
            local pos = rp.Position
            local dist = (pos - originPos).Magnitude
            if dist <= maxObjDist then
                local inFrustum = FrustumCuller:testSphere(pos, 3.0, maxObjDist)
                if inFrustum then
                    local scr, vis = cam:WorldToViewportPoint(pos)
                    if vis and scr.Z > 1 then
                        used = used + 1
                        local slot = objectEspPool[used]
                        registerSapLabel(slot.holder, scr.X, scr.Y, 130, 18, 1)
                        slot.pip.BackgroundColor3 = item.color
                        slot.lbl.TextColor3 = item.color
                        local intDist = math.floor(dist + 0.5)
                        local txt = string.format("%s (%dm)", item.tag, intDist)
                        if slot.lastText ~= txt then
                            slot.lastText = txt
                            slot.lbl.Text = txt
                        end
                        slot.holder.Visible = true
                    end
                end
            end
        end
    end
    for i = used + 1, OBJECT_POOL_SIZE do
        objectEspPool[i].holder.Visible = false
    end
end

local function stepWaypointsESP(cam, originPos)
    for i = 1, #waypoints do
        local wp = waypoints[i]
        if wp and wp.marker then
            if state.waypointEsp then
                local dist = (wp.pos - originPos).Magnitude
                local scr, onScreen = cam:WorldToViewportPoint(wp.pos)
                if onScreen and scr.Z > 1 then
                    registerSapLabel(wp.marker, scr.X, scr.Y, 140, 20, 2)
                    wp.label.Text = string.format("%s  [%dm]", wp.name, math.floor(dist + 0.5))
                    wp.marker.Visible = true
                else
                    wp.marker.Visible = false
                end
            else
                wp.marker.Visible = false
            end
        end
    end
end

local function updateSelfProximityRing(cam, myRoot, closestEnemyDist, closestEnemyArmed)
    if not state.espProximityRing or not myRoot or closestEnemyDist > 55 then
        for i = 1, 16 do hideCachedLine(selfProximityLines[i]) end
        return
    end

    local ringRadius = 18
    local ringColor = Theme.Accent
    if closestEnemyDist <= 18 then
        ringColor = Theme.Danger
    elseif closestEnemyArmed or closestEnemyDist <= 32 then
        ringColor = Theme.WarnAmber
    end

    local groundCenter = myRoot.Position - Vector3.new(0, 2.8, 0)
    for i = 1, 16 do
        local ang = (i - 1) * (math.pi * 2 / 16)
        scratchSelfRingPts[i] = groundCenter + Vector3.new(math.cos(ang) * ringRadius, 0, math.sin(ang) * ringRadius)
    end

    for i = 1, 16 do
        local nxt = (i % 16) + 1
        local okSeg, ax, ay, bx, by = FrustumCuller:projectClippedSegment(scratchSelfRingPts[i], scratchSelfRingPts[nxt])
        if okSeg then
            drawCached2DLine(selfProximityLines[i], ax, ay, bx, by, ringColor, 2, 0.25)
        else
            hideCachedLine(selfProximityLines[i])
        end
    end
end

local function computeEntityBoxBounds(entry, char, hrp, hrpPos, dist, lodTier1Dist, cfgBox3D, accentColor, centerScreen, vpSize, now)
    local boxX, boxY, boxWidth, boxHeight
    if dist <= lodTier1Dist or cfgBox3D then
        local halfExtents = getCachedHalfExtents(entry, now, char)
        local minX, minY = math.huge, math.huge
        local maxX, maxY = -math.huge, -math.huge

        for i = 1, 8 do
            local cOffset = OBB_CORNERS[i] * halfExtents
            local worldPt = hrp.CFrame:PointToWorldSpace(cOffset)
            scratchWorldCorners[i] = worldPt
            local sx, sy = FrustumCuller:projectPointClipped(worldPt)
            scratchProjectedCorners[i] = Vector2.new(sx, sy)
            if sx < minX then minX = sx end
            if sy < minY then minY = sy end
            if sx > maxX then maxX = sx end
            if sy > maxY then maxY = sy end
        end

        boxWidth  = math.clamp(maxX - minX, 14, vpSize.X * 0.65)
        boxHeight = math.clamp(maxY - minY, 18, vpSize.Y * 0.85)
        boxX      = math.floor((minX + maxX) * 0.5 - boxWidth * 0.5)
        boxY      = math.floor((minY + maxY) * 0.5 - boxHeight * 0.5)

        if cfgBox3D and FrameScheduler.tier < 3 and dist <= 240 then
            for i = 1, 12 do
                local edge = OBB_EDGES[i]
                local okSeg, ax, ay, bx, by = FrustumCuller:projectClippedSegment(
                    scratchWorldCorners[edge[1]],
                    scratchWorldCorners[edge[2]]
                )
                if okSeg then
                    drawCached2DLine(entry.box3DLines[i], ax, ay, bx, by, accentColor, 1, 0.25)
                else
                    hideCachedLine(entry.box3DLines[i])
                end
            end
        else
            for i = 1, 12 do hideCachedLine(entry.box3DLines[i]) end
        end
    else
        for i = 1, 12 do hideCachedLine(entry.box3DLines[i]) end
        local _, topY = FrustumCuller:projectPointClipped(hrpPos + Vector3.new(0, 2.85, 0))
        local _, botY = FrustumCuller:projectPointClipped(hrpPos - Vector3.new(0, 3.25, 0))
        boxHeight = math.clamp(math.abs(botY - topY), 16, vpSize.Y * 0.85)
        boxWidth  = math.clamp(boxHeight * 0.60, 12, vpSize.X * 0.60)
        boxX      = math.floor(centerScreen.X - boxWidth * 0.5)
        boxY      = math.floor(topY)
    end
    return boxX, boxY, boxWidth, boxHeight
end

local function renderEntitySkeleton(entry, char, hum, dist, lodTier1Dist, accentColor, now)
    if state.espSkeleton and dist <= (lodTier1Dist + 35) then
        local cachedPairs = getStreamingSafeBoneParts(entry, now, char, hum)
        for i = 1, #entry.boneLines do
            local pairParts = cachedPairs[i]
            local boneLine = entry.boneLines[i]
            if pairParts then
                local partA = pairParts[1]
                local partB = pairParts[2]
                if partA and partB and partA.Parent == char and partB.Parent == char then
                    local okSeg, ax, ay, bx, by = FrustumCuller:projectClippedSegment(partA.Position, partB.Position)
                    if okSeg then
                        drawCached2DLine(boneLine, ax, ay, bx, by, accentColor, 1, 0.2)
                    else
                        hideCachedLine(boneLine)
                    end
                else
                    hideCachedLine(boneLine)
                end
            else
                hideCachedLine(boneLine)
            end
        end
    else
        for i = 1, #entry.boneLines do hideCachedLine(entry.boneLines[i]) end
    end
end

local function renderEntityTrajectories(entry, slot, char, cam, hrpPos, rawVel, smoothVel, futureLeadWorld, flatSpeed, dist, accentColor, dt)
    if state.espVelocityVec and flatSpeed > 3.0 and dist <= 220 then
        local okVec, ax, ay, bx, by = FrustumCuller:projectClippedSegment(hrpPos, futureLeadWorld)
        if okVec then
            drawCached2DLine(entry.velLine, ax, ay, bx, by, Theme.WaypointCyan, 2, 0.15)
            entry.velDot.Position = UDim2.fromOffset(bx, by)
            entry.velDot.Visible = true
        else
            hideCachedLine(entry.velLine)
            entry.velDot.Visible = false
        end
    else
        hideCachedLine(entry.velLine)
        entry.velDot.Visible = false
    end

    if state.espParabolicArc and math.abs(rawVel.Y) > 3.5 and dist <= 200 then
        local myChar = getCharacter()
        losRayParams.FilterDescendantsInstances = myChar and { myChar, char, cam } or { char, cam }
        local validCount, impactWorld = computeGroundClampedBallisticArc(
            hrpPos,
            smoothVel,
            Workspace.Gravity,
            losRayParams,
            FrameScheduler:consumeRayToken()
        )
        for seg = 1, 4 do
            if seg < validCount then
                local okArc, ax, ay, bx, by = FrustumCuller:projectClippedSegment(
                    scratchBallisticPts[seg],
                    scratchBallisticPts[seg + 1]
                )
                if okArc then
                    drawCached2DLine(entry.arcLines[seg], ax, ay, bx, by, Theme.WarnAmber, 2, 0.22)
                else
                    hideCachedLine(entry.arcLines[seg])
                end
            else
                hideCachedLine(entry.arcLines[seg])
            end
        end
        if impactWorld then
            local impX, impY, impVis = FrustumCuller:projectPointClipped(impactWorld)
            if impVis then
                entry.landingDot.Position = UDim2.fromOffset(impX, impY)
                entry.landingDot.Visible = true
            else
                entry.landingDot.Visible = false
            end
        else
            entry.landingDot.Visible = false
        end
    else
        for seg = 1, 4 do hideCachedLine(entry.arcLines[seg]) end
        entry.landingDot.Visible = false
    end

    local base = (slot - 1) * ENTITY_STRIDE
    if state.espMotionTrail and dist <= 185 then
        local headIdx = BufferLib.readu8(entitySoABuffer, base + 56)
        local count   = BufferLib.readu8(entitySoABuffer, base + 57)
        local maxSegs = math.min(count - 1, #entry.trailLines)
        for tIdx = 1, #entry.trailLines do
            if tIdx <= maxSegs then
                local idxA = ((headIdx - tIdx) % TRAIL_CAP) + 1
                local idxB = ((headIdx - tIdx - 1) % TRAIL_CAP) + 1
                local offA = base + TRAIL_BASE_OFFSET + ((idxA - 1) * 12)
                local offB = base + TRAIL_BASE_OFFSET + ((idxB - 1) * 12)
                local pA = Vector3.new(
                    BufferLib.readf32(entitySoABuffer, offA),
                    BufferLib.readf32(entitySoABuffer, offA + 4),
                    BufferLib.readf32(entitySoABuffer, offA + 8)
                )
                local pB = Vector3.new(
                    BufferLib.readf32(entitySoABuffer, offB),
                    BufferLib.readf32(entitySoABuffer, offB + 4),
                    BufferLib.readf32(entitySoABuffer, offB + 8)
                )
                local okTr, ax, ay, bx, by = FrustumCuller:projectClippedSegment(pA, pB)
                if okTr then
                    local fade = math.clamp(0.18 + (tIdx / TRAIL_CAP) * 0.75, 0.18, 0.94)
                    drawCached2DLine(entry.trailLines[tIdx], ax, ay, bx, by, accentColor, 2, fade)
                else
                    hideCachedLine(entry.trailLines[tIdx])
                end
            else
                hideCachedLine(entry.trailLines[tIdx])
            end
        end
    else

    end

        entry.footPulsePhase = (entry.footPulsePhase + dt * 3.2) % 1

                scratchFootPts[s] = footCenter + Vector3.new(math.cos(ang) * ringRadius, 0, math.sin(ang) * ringRadius)
            end

                local okFoot, ax, ay, bx, by = FrustumCuller:projectClippedSegment(scratchFootPts[s], scratchFootPts[nxt])

    currentFocusedTarget = lockedTargetPlayer or bestFocusPlayer

    if state.spectateTarget and cam.CameraType == Enum.CameraType.Custom then
        local targetToWatch = lockedTargetPlayer or currentFocusedTarget

            end
        else

                cam.CameraSubject = myHum
            end
        end
    end
end

local SpringEngine = {
    activeSprings = {},
}

function SpringEngine:bind1D(initialVal, omega, onStep)
    local sp = {
        x      = initialVal,
        v      = 0,
        target = initialVal,
        omega  = omega or 22.0,
        awake  = false,
        onStep = onStep,
    }
    function sp:setGoal(goal, impulseVel)
        self.target = goal
        if impulseVel then self.v = self.v + impulseVel end
        if not self.awake then
            self.awake = true
            table.insert(SpringEngine.activeSprings, self)
        end
    end
    return sp
end

function SpringEngine:stepAll(dt)
    local count = #self.activeSprings
    if count == 0 then return end
    local clampedDt = math.clamp(dt, 0.001, 0.25)
    local i = 1
    while i <= count do
        local s = self.activeSprings[i]
        local w = s.omega
        local decay = math.exp(-w * clampedDt)
        local c1 = s.x - s.target
        local c2 = s.v + w * c1
        s.x = s.target + (c1 + c2 * clampedDt) * decay
        s.v = (c2 - w * (c1 + c2 * clampedDt)) * decay
        s.onStep(s.x)

        if math.abs(s.x - s.target) < 0.002 and math.abs(s.v) < 0.01 then
            s.x = s.target
            s.v = 0
            s.onStep(s.target)
            s.awake = false
            self.activeSprings[i] = self.activeSprings[count]
            self.activeSprings[count] = nil
            count = count - 1
        else
            i = i + 1
        end
    end
end

local mainScaleRef = nil

local function clampGuiToViewport(obj)
    local currentCamera = Workspace.CurrentCamera
    if not currentCamera or not obj.Visible then return end

    local size = currentCamera.ViewportSize
    if size.X < 64 or size.Y < 64 then return end

    local scaleFactor = (obj.Name == "MainFrame" and mainScaleRef and mainScaleRef.Scale > 0.05) and mainScaleRef.Scale or 1.0
    local unscaledW = obj.AbsoluteSize.X / scaleFactor
    local unscaledH = obj.AbsoluteSize.Y / scaleFactor
    local margin = 8
    local anchor = obj.AnchorPoint
    local minX = margin + unscaledW * anchor.X
    local maxX = size.X - margin - unscaledW * (1 - anchor.X)
    local minY = margin + unscaledH * anchor.Y
    local maxY = size.Y - margin - unscaledH * (1 - anchor.Y)
    local position = obj.Position

    local lowX  = minX - position.X.Scale * size.X
    local highX = math.max(lowX, maxX - position.X.Scale * size.X)
    local lowY  = minY - position.Y.Scale * size.Y
    local highY = math.max(lowY, maxY - position.Y.Scale * size.Y)

    obj.Position = UDim2.new(
        position.X.Scale,
        math.clamp(position.X.Offset, lowX, highX),
        position.Y.Scale,
        math.clamp(position.Y.Offset, lowY, highY)
    )
end

local function makeDraggable(targetObj, handleObj, onClickCallback)
    handleObj = handleObj or targetObj
    local dragStartPos = nil
    local wasDragged = false

    table.insert(connections, handleObj.InputBegan:Connect(function(input)
        if activeSliderDrag then return end
        if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
            dragStartPos = input.Position
            wasDragged = false
            activeDrag = {
                obj = targetObj,
                input = input,
                startPos = targetObj.Position,
                dragStart = input.Position,
                markDragged = function()
                    wasDragged = true
                end,
            }
        end
    end))

    if type(onClickCallback) == "function" then
        table.insert(connections, handleObj.InputEnded:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.MouseButton1
            or input.UserInputType == Enum.UserInputType.Touch then
                local movedDist = dragStartPos and (input.Position - dragStartPos).Magnitude or 0
                if not wasDragged and movedDist < 8 then
                    onClickCallback()
                end
            end
        end))
    end
end

        activeDrag.startPos.X.Scale, targetX,

    )
end))

local function updateResponsiveLayout()
    local currentCamera = Workspace.CurrentCamera
    if not currentCamera then return end

    viewport = currentCamera.ViewportSize
    shortestSide = math.min(viewport.X, viewport.Y)
    deviceType = UserInputService.TouchEnabled
        and (shortestSide >= 600 and "Tablet" or "Phone")
        or "Desktop"

    mainFrame.Size = getGuiSize()

    task.defer(function()
        if not closed then
            clampGuiToViewport(mainFrame)
            clampGuiToViewport(miniFrame)
        end
    end)

    if deviceType == "Phone" then
        miniFrame.Size = UDim2.fromOffset(42, 42)
    elseif deviceType == "Tablet" then
        miniFrame.Size = UDim2.fromOffset(48, 48)
    else
        miniFrame.Size = UDim2.fromOffset(46, 46)
    end
end

local function watchCamera(cameraToWatch)
    if viewportConnection then
        viewportConnection:Disconnect()
        viewportConnection = nil
    end
    if cameraToWatch then
        viewportConnection = cameraToWatch:GetPropertyChangedSignal("ViewportSize"):Connect(updateResponsiveLayout)
        table.insert(connections, viewportConnection)
    end
end
watchCamera(Workspace.CurrentCamera)
table.insert(connections, Workspace:GetPropertyChangedSignal("CurrentCamera"):Connect(function()
    watchCamera(Workspace.CurrentCamera)
    updateResponsiveLayout()
    if state.fovEnabled and fovLock then
        task.defer(function() fovLock:install() end)
    end
end))
updateResponsiveLayout()

        local factor = 10 ^ decimals
        local clamped = math.clamp(raw, minVal, maxVal)

        sliderFill.Size = UDim2.fromScale(alpha, 1)
        valuePill.Text = formatVal(currentVal)
        if notify and onValue then
            pcall(onValue, currentVal)

        animateSwitch(on)

        local success, err = pcall(onToggle, on)
        if not success then
            warn("[Johan Hub] Failed to toggle " .. text .. ": " .. tostring(err))
            isOn = not on
            animateSwitch(isOn)
            pcall(onToggle, isOn)
            return false
        end
        return true
    end

    end))

    table.insert(connections, card.MouseLeave:Connect(function()

    end))

    return api
end

local CONFIG_FILE = string.format("JohanHub_v6_%d.json", game.PlaceId)

local function saveProfileToDisk(badge)
    if type(writefile) ~= "function" then
        if badge then badge.Text = "NO FS" end
        return
    end
    local wpList = {}
    for i = 1, #waypoints do
        local w = waypoints[i]
        wpList[i] = { name = w.name, x = w.pos.X, y = w.pos.Y, z = w.pos.Z }
    end
    local togglesData = {}
    for k, api in pairs(toggleRegistry) do
        togglesData[k] = {
            on  = api.isOn and api.isOn() or false,
            val = api.getValue and api.getValue() or nil,
        }
    end
    local payload = {
        theme     = state.themeName,
        toggles   = togglesData,
        waypoints = wpList,
    }
    local ok = pcall(function()
        writefile(CONFIG_FILE, HttpService:JSONEncode(payload))
    end)
    if badge then
        badge.Text = ok and "SAVED!" or "ERR"
        task.delay(1.2, function()
            if badge and badge.Parent then badge.Text = "SAVE" end
        end)
    end
end

local function loadProfileFromDisk(badge)
    if type(isfile) ~= "function" or type(readfile) ~= "function" or not isfile(CONFIG_FILE) then
        if badge then
            badge.Text = "NONE"
            task.delay(1.2, function()
                if badge and badge.Parent then badge.Text = "LOAD" end
            end)
        end
        return
    end
    local ok, data = pcall(function()
        return HttpService:JSONDecode(readfile(CONFIG_FILE))
    end)
    if not ok or type(data) ~= "table" then return end

    if type(data.theme) == "string" and ThemePalettes[data.theme] then
        applyThemePalette(data.theme)
        local foundIdx = table.find(themeOrder, data.theme)
        if foundIdx then currentThemeIdx = foundIdx end
        if themeCycleBadge then themeCycleBadge.Text = data.theme end
    end

    if type(data.toggles) == "table" then
        for k, saved in pairs(data.toggles) do
            local api = toggleRegistry[k]
            if api then
                if saved.val ~= nil and api.setValue then
                    pcall(api.setValue, saved.val)
                end
                if type(saved.on) == "boolean" and api.setState then
                    pcall(api.setState, saved.on)
                end
            end
        end
    end

    if type(data.waypoints) == "table" then
        for _, wp in ipairs(waypoints) do
            if wp.marker then pcall(function() wp.marker:Destroy() end) end
            if wp.card then pcall(function() wp.card:Destroy() end) end
        end
        table.clear(waypoints)
        for _, w in ipairs(data.waypoints) do
            if type(w) == "table" and w.x and w.y and w.z then
                addWaypointBookmark(w.name, Vector3.new(w.x, w.y, w.z))
            end
        end
    end

    if badge then
        badge.Text = "LOADED!"
        task.delay(1.2, function()
            if badge and badge.Parent then badge.Text = "LOAD" end
        end)
    end
end

    local radiusPx    = 78

    for i = 1, numSectors do
        local ang = (i - 1) * (math.pi * 2 / numSectors) - (math.pi * 0.5)
        local ox  = math.cos(ang) * radiusPx
        local oy  = math.sin(ang) * radiusPx

        local card = Instance.new("Frame")
        card.AnchorPoint      = Vector2.new(0.5, 0.5)
        card.Position         = UDim2.new(0.5, math.floor(ox + 0.5), 0.5, math.floor(oy + 0.5))

                sectorCards[i].card.BackgroundColor3 = Theme.BgRow
                sectorCards[i].stroke.Color = Theme.Stroke
            end
            return
        end

        local angle = math.atan2(delta.Y, delta.X) + (math.pi * 0.5)
        local step  = (math.pi * 2) / numSectors
        local norm  = (angle + step * 0.5) % (math.pi * 2)
        local idx   = (math.floor(norm / step) % numSectors) + 1

        end
    end

    return api

local EventBus = {
    topics = {},
}

function EventBus:Subscribe(topic, callback, priority)
    local list = self.topics[topic]
    if not list then
        list = {}
        self.topics[topic] = list
    end
    local sub = { fn = callback, priority = priority or 0, connected = true }
    table.insert(list, sub)
    table.sort(list, function(a, b) return a.priority > b.priority end)

    return {
        Disconnect = function()
            if not sub.connected then return end
            sub.connected = false
            local idx = table.find(list, sub)
            if idx then table.remove(list, idx) end
        end,
    }
end

function EventBus:Publish(topic, ...)
    local list = self.topics[topic]
    if not list then return end
    for i = 1, #list do
        local sub = list[i]
        if sub and sub.connected then
            local ok, err = pcall(sub.fn, ...)
            if not ok then
                warn("[JohanHub EventBus] Error in topic '" .. topic .. "': " .. tostring(err))
            end
        end
    end
end

function EventBus:ClearAll()
    table.clear(self.topics)
end

local GameProfileRegistry = {
    profiles      = {},
    activeProfile = nil,
}

function GameProfileRegistry:Register(profileDef)
    table.insert(self.profiles, profileDef)
end

function GameProfileRegistry:AutoMount(hubContext)
end
    local placeId = game.PlaceId
    local gameId  = game.GameId

    for i = 1, #self.profiles do
        local prof = self.profiles[i]
        local matchPlace = prof.PlaceIds and table.find(prof.PlaceIds, placeId) ~= nil
        local matchGame  = prof.GameIds and table.find(prof.GameIds, gameId) ~= nil
        if matchPlace or matchGame then
            self.activeProfile = prof
            if prof.WhitelistObjects and hubContext.Scanner then
                for _, token in ipairs(prof.WhitelistObjects) do
                    hubContext.Scanner.whitelist[string.lower(token)] = true
                end
            end
            if type(prof.OnMount) == "function" then
                pcall(prof.OnMount, hubContext, EventBus)
            end
            EventBus:Publish("OnProfileMounted", prof.Name)
            break
        end
    end
end
